===== backend\app\Models\Scopes\TenantScope.php =====

<?php

namespace App\Models\Scopes;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Scope;
use Illuminate\Support\Facades\Auth;

class TenantScope implements Scope
{
    public function apply(Builder $builder, Model $model): void
    {
        $user = Auth::user();

        if ($user !== null && $user->company_id !== null) {
            $builder->where(
                $model->qualifyColumn('company_id'),
                $user->company_id
            );
        }
    }
}

<?php

namespace App\Http\Controllers;

use App\Http\Requests\IndexInternalRequestRequest;
use App\Http\Requests\StoreInternalRequestRequest;
use App\Http\Requests\UpdateInternalRequestRequest;
use App\Http\Resources\InternalRequestResource;
use App\Models\InternalRequest;
use App\Services\InternalRequestService;
use Illuminate\Http\Resources\Json\AnonymousResourceCollection;
use Illuminate\Http\Response;
use Illuminate\Support\Facades\Gate;

final class InternalRequestController extends Controller
{
    public function __construct(
        private readonly InternalRequestService $service
    ) {}

    public function index(
        IndexInternalRequestRequest $request
    ): AnonymousResourceCollection {
        Gate::authorize(
            'viewAny',
            InternalRequest::class
        );

        return InternalRequestResource::collection(
            $this->service->paginate(
                $request->validated(),
                $request->user()
            )
        );
    }

    public function store(
        StoreInternalRequestRequest $request
    ): InternalRequestResource {
        Gate::authorize(
            'create',
            InternalRequest::class
        );

        return new InternalRequestResource(
            $this->service->createDraft(
                $request->validated(),
                $request->user()
            )
        );
    }

    public function show(
        InternalRequest $internalRequest
    ): InternalRequestResource {
        Gate::authorize('view', $internalRequest);

        return new InternalRequestResource(
            $internalRequest->load([
                'requestType:id,name,code',
                'requesterEmployee:id,first_name,last_name,employee_number',
                'workflowDefinition:id,name,code,version',
                'currentWorkflowStep:id,name,code,step_order',

                'workflowInstance.tasks.workflowStep:id,name,code,step_order',

                'workflowInstance.tasks.assignedEmployee:id,first_name,last_name,employee_number',
            ])
        );
    }

    public function update(
        UpdateInternalRequestRequest $request,
        InternalRequest $internalRequest
    ): InternalRequestResource {
        Gate::authorize('update', $internalRequest);

        return new InternalRequestResource(
            $this->service->updateDraft(
                $internalRequest,
                $request->validated()
            )
        );
    }

    public function destroy(
        InternalRequest $internalRequest
    ): Response {
        Gate::authorize('delete', $internalRequest);

        $this->service->deleteDraft($internalRequest);

        return response()->noContent();
    }
}

===== backend\app\Services\InternalRequestService.php =====

<?php

namespace App\Services;

use App\Models\Employee;
use App\Models\InternalRequest;
use App\Models\RequestType;
use App\Models\User;
use App\Models\WorkflowInstance;
use App\Models\WorkflowTask;
use Illuminate\Auth\Access\AuthorizationException;
use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Database\Eloquent\Builder;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

final class InternalRequestService
{
    public function __construct(
        private readonly WorkflowAssignmentResolver $resolver
    ) {}

    public function paginate(
        array $filters,
        User $user
    ): LengthAwarePaginator {
        $query = InternalRequest::query()
            ->with([
                'requestType:id,name,code',
                'requesterEmployee:id,first_name,last_name,employee_number',
                'workflowDefinition:id,name,code,version',
                'currentWorkflowStep:id,name,code,step_order',
            ]);

        if (! $user->can('requests.view_all')) {
            $query->whereHas(
                'requesterEmployee',
                fn (Builder $employeeQuery) => $employeeQuery->where(
                    'user_id',
                    $user->id
                )
            );
        }

        $search = mb_strtolower(
            trim((string) ($filters['search'] ?? ''))
        );

        if ($search !== '') {
            $likeSearch = "%{$search}%";

            $query->where(
                function (Builder $searchQuery) use ($likeSearch): void {
                    $searchQuery
                        ->whereRaw(
                            'LOWER(reference_number) LIKE ?',
                            [$likeSearch]
                        )
                        ->orWhereRaw(
                            'LOWER(title) LIKE ?',
                            [$likeSearch]
                        )
                        ->orWhereRaw(
                            "LOWER(COALESCE(description, '')) LIKE ?",
                            [$likeSearch]
                        );
                }
            );
        }

        $status = (string) ($filters['status'] ?? 'all');

        if ($status !== 'all') {
            $query->where('status', $status);
        }

        $priority = (string) (
            $filters['priority'] ?? 'all'
        );

        if ($priority !== 'all') {
            $query->where('priority', $priority);
        }

        if (! empty($filters['request_type_id'])) {
            $query->where(
                'request_type_id',
                (int) $filters['request_type_id']
            );
        }

        return $query
            ->orderByDesc('created_at')
            ->orderByDesc('id')
            ->paginate(
                min(
                    max(
                        (int) ($filters['per_page'] ?? 15),
                        5
                    ),
                    100
                )
            )
            ->appends($filters);
    }

    public function createDraft(
        array $data,
        User $user
    ): InternalRequest {
        $requester = $this->resolveRequester(
            $data['requester_employee_id'] ?? null,
            $user
        );

        $requestType = RequestType::query()
            ->with('workflowDefinition')
            ->findOrFail($data['request_type_id']);

        if (! $requestType->is_active) {
            throw ValidationException::withMessages([
                'request_type_id' => [
                    'The selected request type is inactive.',
                ],
            ]);
        }

        if (! $requestType->workflowDefinition->is_active) {
            throw ValidationException::withMessages([
                'request_type_id' => [
                    'The selected request type uses an inactive workflow.',
                ],
            ]);
        }

        unset($data['requester_employee_id']);

        return InternalRequest::query()
            ->create([
                ...$data,
                'company_id' => (int) $user->company_id,
                'request_type_id' => $requestType->id,
                'workflow_definition_id' => $requestType->workflow_definition_id,
                'requester_employee_id' => $requester->id,
                'reference_number' => $this->generateReferenceNumber(
                    (int) $user->company_id
                ),
                'status' => 'draft',
            ])
            ->load([
                'requestType:id,name,code',
                'requesterEmployee:id,first_name,last_name,employee_number',
                'workflowDefinition:id,name,code,version',
                'currentWorkflowStep:id,name,code,step_order',
            ]);
    }

    public function updateDraft(
        InternalRequest $internalRequest,
        array $data
    ): InternalRequest {
        if ($internalRequest->status !== 'draft') {
            throw ValidationException::withMessages([
                'status' => [
                    'Only draft requests can be updated.',
                ],
            ]);
        }

        $internalRequest->update($data);

        return $internalRequest
            ->fresh()
            ->load([
                'requestType:id,name,code',
                'requesterEmployee:id,first_name,last_name,employee_number',
                'workflowDefinition:id,name,code,version',
                'currentWorkflowStep:id,name,code,step_order',
            ]);
    }

    public function submit(
        InternalRequest $internalRequest
    ): InternalRequest {
        return DB::transaction(
            function () use ($internalRequest): InternalRequest {
                $lockedRequest = InternalRequest::query()
                    ->whereKey($internalRequest->id)
                    ->lockForUpdate()
                    ->firstOrFail();

                if ($lockedRequest->status !== 'draft') {
                    throw ValidationException::withMessages([
                        'status' => [
                            'Only draft requests can be submitted.',
                        ],
                    ]);
                }

                $lockedRequest->load([
                    'requesterEmployee',
                    'workflowDefinition.steps',
                ]);

                $workflowDefinition =
                    $lockedRequest->workflowDefinition;

                if (! $workflowDefinition->is_active) {
                    throw ValidationException::withMessages([
                        'workflow' => [
                            'The workflow is inactive.',
                        ],
                    ]);
                }

                $firstStep = $workflowDefinition
                    ->steps
                    ->sortBy('step_order')
                    ->first();

                if ($firstStep === null) {
                    throw ValidationException::withMessages([
                        'workflow' => [
                            'The workflow does not contain approval steps.',
                        ],
                    ]);
                }

                $assignment = $this->resolver->resolve(
                    $firstStep,
                    $lockedRequest->requesterEmployee
                );

                $now = now();

                $workflowInstance =
                    WorkflowInstance::query()->create([
                        'company_id' => $lockedRequest->company_id,
                        'internal_request_id' => $lockedRequest->id,
                        'workflow_definition_id' => $workflowDefinition->id,
                        'current_workflow_step_id' => $firstStep->id,
                        'status' => 'active',
                        'started_at' => $now,
                    ]);

                WorkflowTask::query()->create([
                    'company_id' => $lockedRequest->company_id,
                    'workflow_instance_id' => $workflowInstance->id,
                    'workflow_step_id' => $firstStep->id,
                    'assigned_employee_id' => $assignment[
                            'assigned_employee_id'
                        ],
                    'assigned_role' => $assignment['assigned_role'],
                    'status' => 'pending',
                    'due_at' => $firstStep->sla_hours
                        ? $now->copy()->addHours(
                            $firstStep->sla_hours
                        )
                        : null,
                ]);

                $lockedRequest->update([
                    'status' => 'in_review',
                    'current_workflow_step_id' => $firstStep->id,
                    'submitted_at' => $now,
                ]);

                return $lockedRequest
                    ->fresh()
                    ->load([
                        'requestType:id,name,code',
                        'requesterEmployee:id,first_name,last_name,employee_number',
                        'workflowDefinition:id,name,code,version',
                        'currentWorkflowStep:id,name,code,step_order',

                        'workflowInstance.tasks.workflowStep:id,name,code,step_order',

                        'workflowInstance.tasks.assignedEmployee:id,first_name,last_name,employee_number',
                    ]);
            }
        );
    }

    public function deleteDraft(
        InternalRequest $internalRequest
    ): void {
        if ($internalRequest->status !== 'draft') {
            throw ValidationException::withMessages([
                'status' => [
                    'Only draft requests can be deleted.',
                ],
            ]);
        }

        $internalRequest->delete();
    }

    private function resolveRequester(
        ?int $requestedEmployeeId,
        User $user
    ): Employee {
        if ($requestedEmployeeId !== null) {
            $requester = Employee::query()
                ->findOrFail($requestedEmployeeId);

            $isOwnEmployee =
                $user->employee?->id === $requester->id;

            if (
                ! $isOwnEmployee
                && ! $user->can(
                    'requests.create_for_others'
                )
            ) {
                throw new AuthorizationException;
            }

            return $requester;
        }

        $requester = $user->employee;

        if ($requester === null) {
            throw ValidationException::withMessages([
                'requester_employee_id' => [
                    'This user is not linked to an employee record.',
                ],
            ]);
        }

        return $requester;
    }

    private function generateReferenceNumber(
        int $companyId
    ): string {
        do {
            $reference = sprintf(
                'REQ-%s-%s',
                now()->format('Ymd'),
                Str::upper(Str::random(8))
            );
        } while (
            InternalRequest::query()
                ->where('company_id', $companyId)
                ->where(
                    'reference_number',
                    $reference
                )
                ->exists()
        );

        return $reference;
    }
}

<?php

namespace App\Services;

use App\Models\InternalRequest;
use App\Models\User;
use App\Models\WorkflowInstance;
use App\Models\WorkflowStep;
use App\Models\WorkflowTask;
use App\Models\WorkflowTaskDecision;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

final class ApprovalActionService
{
    public function __construct(
        private readonly WorkflowAssignmentResolver $resolver,
        private readonly ApprovalEligibilityService $eligibility,
        private readonly WorkflowTaskService $taskService
    ) {}

    public function approve(WorkflowTask $workflowTask, User $actor, ?string $note): WorkflowTask
    {
        return DB::transaction(function () use ($workflowTask, $actor, $note): WorkflowTask {
            [$task, $instance, $request, $step] = $this->lockRuntime($workflowTask);
            $this->ensurePendingRuntime($task, $instance, $request);
            $this->recordDecision($task, $actor, 'approved', $note);

            $approvalCount = WorkflowTaskDecision::query()
                ->where('workflow_task_id', $task->id)
                ->where('decision', 'approved')
                ->count();

            if ($approvalCount < $this->requiredApprovals($task, $step)) {
                return $this->taskService->loadForDisplay($task->fresh());
            }

            $now = now();

            $task->update([
                'status' => 'approved',
                'acted_at' => $now,
                'acted_by_employee_id' => $actor->employee?->id,
                'decision_note' => $note,
            ]);

            $nextStep = WorkflowStep::query()
                ->where('workflow_definition_id', $instance->workflow_definition_id)
                ->where('step_order', '>', $step->step_order)
                ->orderBy('step_order')
                ->first();

            if ($step->is_final || $nextStep === null) {
                $instance->update(['status' => 'completed', 'completed_at' => $now]);
                $request->update(['status' => 'approved', 'completed_at' => $now]);

                return $this->taskService->loadForDisplay($task->fresh());
            }

            $request->load('requesterEmployee');
            $assignment = $this->resolver->resolve($nextStep, $request->requesterEmployee);

            WorkflowTask::query()->create([
                'company_id' => $request->company_id,
                'workflow_instance_id' => $instance->id,
                'workflow_step_id' => $nextStep->id,
                'assigned_employee_id' => $assignment['assigned_employee_id'],
                'assigned_role' => $assignment['assigned_role'],
                'status' => 'pending',
                'due_at' => $nextStep->sla_hours
                    ? $now->copy()->addHours($nextStep->sla_hours)
                    : null,
            ]);

            $instance->update(['current_workflow_step_id' => $nextStep->id]);
            $request->update(['current_workflow_step_id' => $nextStep->id]);

            return $this->taskService->loadForDisplay($task->fresh());
        });
    }

    public function reject(WorkflowTask $workflowTask, User $actor, string $note): WorkflowTask
    {
        return DB::transaction(function () use ($workflowTask, $actor, $note): WorkflowTask {
            [$task, $instance, $request, $step] = $this->lockRuntime($workflowTask);
            $this->ensurePendingRuntime($task, $instance, $request);

            if (! $step->can_reject) {
                throw ValidationException::withMessages([
                    'decision' => ['This workflow step does not allow rejection.'],
                ]);
            }

            $this->recordDecision($task, $actor, 'rejected', $note);
            $now = now();

            $task->update([
                'status' => 'rejected',
                'acted_at' => $now,
                'acted_by_employee_id' => $actor->employee?->id,
                'decision_note' => $note,
            ]);

            $instance->update(['status' => 'rejected', 'completed_at' => $now]);
            $request->update(['status' => 'rejected', 'completed_at' => $now]);

            return $this->taskService->loadForDisplay($task->fresh());
        });
    }

    private function lockRuntime(WorkflowTask $workflowTask): array
    {
        $task = WorkflowTask::query()->whereKey($workflowTask->id)->lockForUpdate()->firstOrFail();
        $instance = WorkflowInstance::query()->whereKey($task->workflow_instance_id)->lockForUpdate()->firstOrFail();
        $request = InternalRequest::query()->whereKey($instance->internal_request_id)->lockForUpdate()->firstOrFail();
        $step = WorkflowStep::query()->findOrFail($task->workflow_step_id);

        return [$task, $instance, $request, $step];
    }

    private function ensurePendingRuntime(
        WorkflowTask $task,
        WorkflowInstance $instance,
        InternalRequest $request
    ): void {
        if ($task->status !== 'pending' || $instance->status !== 'active' || $request->status !== 'in_review') {
            throw ValidationException::withMessages([
                'status' => ['This approval task is no longer pending.'],
            ]);
        }
    }

    private function recordDecision(
        WorkflowTask $task,
        User $actor,
        string $decision,
        ?string $note
    ): void {
        $alreadyActed = WorkflowTaskDecision::query()
            ->where('workflow_task_id', $task->id)
            ->where('acted_by_user_id', $actor->id)
            ->exists();

        if ($alreadyActed) {
            throw ValidationException::withMessages([
                'decision' => ['You have already acted on this approval task.'],
            ]);
        }

        WorkflowTaskDecision::query()->create([
            'company_id' => $task->company_id,
            'workflow_task_id' => $task->id,
            'acted_by_user_id' => $actor->id,
            'acted_by_employee_id' => $actor->employee?->id,
            'decision' => $decision,
            'note' => $note,
            'acted_at' => now(),
        ]);
    }

    private function requiredApprovals(WorkflowTask $task, WorkflowStep $step): int
    {
        $eligible = $this->eligibility->eligibleUserCount($task);

        if ($step->approval_mode === 'all') {
            return max(1, $eligible);
        }

        $configured = max(1, $step->required_approvals);

        return $eligible <= 0 ? $configured : min($configured, $eligible);
    }
}

function extractErrorMessage(error: unknown): string {
  const axiosError =
    error as AxiosError<LaravelErrorResponse>

  const validationErrors =
    axiosError.response?.data?.errors

  if (validationErrors) {
    const firstMessage = Object.values(
      validationErrors,
    ).flat()[0]

    if (firstMessage) {
      return firstMessage
    }
  }

  return (
    axiosError.response?.data?.message ??
    'The operation could not be completed.'
  )
}

function formatDateTime(value: string | null): string {
  if (!value) {
    return 'Not set'
  }

  const date = new Date(value)

  if (Number.isNaN(date.getTime())) {
    return 'Not set'
  }

  return new Intl.DateTimeFormat('en-GB', {
    dateStyle: 'medium',
    timeStyle: 'short',
  }).format(date)
}

function statusClasses(
  status: InternalRequestStatus,
): string {
  if (status === 'approved') {
    return 'border-emerald-400/20 bg-emerald-500/10 text-emerald-300'
  }

  if (status === 'rejected') {
    return 'border-rose-400/20 bg-rose-500/10 text-rose-300'
  }

  if (status === 'in_review') {
    return 'border-sky-400/20 bg-sky-500/10 text-sky-300'
  }

  if (status === 'cancelled') {
    return 'border-slate-700 bg-slate-900 text-slate-500'
  }

  return 'border-amber-400/20 bg-amber-500/10 text-amber-300'
}

function RequestDialog({
  request,
  requestTypes,
  employees,
  canCreateForOthers,
  isSubmitting,
  serverError,
  onClose,
  onSubmit,
}: RequestDialogProps) {
  const [form, setForm] =
    useState<InternalRequestFormValues>(
      emptyForm,
    )

  const [errors, setErrors] = useState<
    Partial<
      Record<
        keyof InternalRequestFormValues,
        string
      >
    >
  >({})

  useEffect(() => {
    if (request) {
      setForm({
        request_type_id:
          request.request_type_id.toString(),
        requester_employee_id:
          request.requester_employee_id.toString(),
        title: request.title,
        description: request.description ?? '',
        priority: request.priority,
      })
    } else {
      setForm(emptyForm)
    }

    setErrors({})
  }, [request])

  const handleSubmit = (
    event: FormEvent<HTMLFormElement>,
  ) => {
    event.preventDefault()

    const result =
      internalRequestSchema.safeParse(form)

    if (!result.success) {
      const nextErrors: Partial<
        Record<
          keyof InternalRequestFormValues,
          string
        >
      > = {}

      for (const issue of result.error.issues) {
        const field = issue.path[0]

        if (
          typeof field === 'string' &&
          nextErrors[
            field as keyof InternalRequestFormValues
          ] === undefined
        ) {
          nextErrors[
            field as keyof InternalRequestFormValues
          ] = issue.message
        }
      }

      setErrors(nextErrors)

      return
    }

    setErrors({})

    if (request) {
      onSubmit({
        title: result.data.title,
        description: nullableString(
          result.data.description,
        ),
        payload: request.payload,
        priority: result.data.priority,
      })

      return
    }

    onSubmit({
      request_type_id: Number(
        result.data.request_type_id,
      ),

      requester_employee_id:
        result.data.requester_employee_id === ''
          ? null
          : Number(
              result.data.requester_employee_id,
            ),

      title: result.data.title,

      description: nullableString(
        result.data.description,
      ),

      payload: null,
      priority: result.data.priority,
    })
  }

  const fieldClass =
    'h-11 w-full rounded-xl border border-slate-800 bg-slate-950 px-3 text-sm text-slate-200 outline-none transition placeholder:text-slate-700 focus:border-sky-400/40 disabled:opacity-50'

  return (
    <div className="fixed inset-0 z-50 overflow-y-auto bg-slate-950/80 p-4 backdrop-blur-sm">
      <div className="mx-auto my-4 w-full max-w-2xl overflow-hidden rounded-3xl border border-slate-700/80 bg-[#07101f] shadow-2xl shadow-slate-950">
        <div className="flex items-center justify-between border-b border-slate-800 p-6">
          <div>
            <p className="text-xs uppercase tracking-[0.16em] text-sky-400">
              Internal request
            </p>

            <h2 className="mt-2 text-xl font-semibold">
              {request
                ? 'Edit draft'
                : 'Create request draft'}
            </h2>
          </div>

          <button
            type="button"
            onClick={onClose}
            disabled={isSubmitting}
            className="grid h-10 w-10 place-items-center rounded-xl border border-slate-800 text-slate-500 hover:text-slate-200 disabled:opacity-40"
          >
            <X size={18} />
          </button>
        </div>

        <form
          onSubmit={handleSubmit}
          className="space-y-5 p-6"
        >
          {serverError && (
            <div className="flex items-start gap-3 rounded-2xl border border-rose-400/20 bg-rose-500/10 p-4 text-sm text-rose-200">
              <AlertTriangle
                size={18}
                className="mt-0.5 shrink-0"
              />
              <span>{serverError}</span>
            </div>
          )}

          <label>
            <span className="mb-2 block text-xs text-slate-500">
              Request type
            </span>

            <select
              value={form.request_type_id}
              onChange={(event) =>
                setForm((current) => ({
                  ...current,
                  request_type_id:
                    event.target.value,
                }))
              }
              disabled={request !== null}
              className={fieldClass}
            >
              <option value="">
                Select request type
              </option>

              {requestTypes.map((requestType) => (
                <option
                  key={requestType.id}
                  value={requestType.id}
                >
                  {requestType.name} (
                  {requestType.code})
                </option>
              ))}
            </select>

            {errors.request_type_id && (
              <p className="mt-2 text-xs text-rose-300">
                {errors.request_type_id}
              </p>
            )}
          </label>

          {canCreateForOthers && (
            <label>
              <span className="mb-2 block text-xs text-slate-500">
                Requester employee
              </span>

              <select
                value={form.requester_employee_id}
                onChange={(event) =>
                  setForm((current) => ({
                    ...current,
                    requester_employee_id:
                      event.target.value,
                  }))
                }
                disabled={request !== null}
                className={fieldClass}
              >
                <option value="">
                  Use my linked employee record
                </option>

                {employees.map((employee) => (
                  <option
                    key={employee.id}
                    value={employee.id}
                  >
                    {employee.full_name} -{' '}
                    {employee.employee_number}
                  </option>
                ))}
              </select>
            </label>
          )}

          <label>
            <span className="mb-2 block text-xs text-slate-500">
              Title
            </span>

            <input
              value={form.title}
              onChange={(event) =>
                setForm((current) => ({
                  ...current,
                  title: event.target.value,
                }))
              }
              placeholder="Annual leave for three days"
              className={fieldClass}
            />

            {errors.title && (
              <p className="mt-2 text-xs text-rose-300">
                {errors.title}
              </p>
            )}
          </label>

          <label>
            <span className="mb-2 block text-xs text-slate-500">
              Description
            </span>

            <textarea
              rows={5}
              value={form.description}
              onChange={(event) =>
                setForm((current) => ({
                  ...current,
                  description: event.target.value,
                }))
              }
              className="w-full resize-none rounded-xl border border-slate-800 bg-slate-950 px-3 py-3 text-sm leading-6 text-slate-200 outline-none transition focus:border-sky-400/40"
            />
          </label>

          <label>
            <span className="mb-2 block text-xs text-slate-500">
              Priority
            </span>

            <select
              value={form.priority}
              onChange={(event) =>
                setForm((current) => ({
                  ...current,
                  priority:
                    event.target.value as RequestPriority,
                }))
              }
              className={fieldClass}
            >
              <option value="low">Low</option>
              <option value="normal">Normal</option>
              <option value="high">High</option>
              <option value="urgent">Urgent</option>
            </select>
          </label>

          <div className="flex justify-end gap-3 border-t border-slate-800 pt-5">
            <button
              type="button"
              onClick={onClose}
              disabled={isSubmitting}
              className="h-11 rounded-xl border border-slate-700 px-5 text-sm text-slate-400 disabled:opacity-40"
            >
              Cancel
            </button>

            <button
              type="submit"
              disabled={isSubmitting}
              className="flex h-11 items-center gap-2 rounded-xl bg-sky-500 px-5 text-sm font-semibold text-slate-950 hover:bg-sky-400 disabled:opacity-60"
            >
              {isSubmitting && (
                <LoaderCircle
                  size={16}
                  className="animate-spin"
                />
              )}

              {request
                ? 'Save draft'
                : 'Create draft'}
            </button>
          </div>
        </form>
      </div>
    </div>
  )
}

function InternalRequestsPage() {
  const navigate = useNavigate()
  const queryClient = useQueryClient()

  const [draftFilters, setDraftFilters] =
    useState<InternalRequestFilters>(
      initialFilters,
    )

  const [filters, setFilters] =
    useState<InternalRequestFilters>(
      initialFilters,
    )

  const [selectedRequestId, setSelectedRequestId] =
    useState<number | null>(null)

  const [dialogOpen, setDialogOpen] =
    useState(false)

  const [editingRequest, setEditingRequest] =
    useState<InternalRequest | null>(null)

  const [mutationError, setMutationError] =
    useState<string | null>(null)

  const {
    data: user,
    isPending: isUserPending,
    isError: isUserError,
  } = useQuery({
    queryKey: ['current-user'],
    queryFn: authService.getCurrentUser,
    retry: false,
  })

  const canView =
    user?.permissions?.includes(
      'requests.view',
    ) ?? false

  const canCreate =
    user?.permissions?.includes(
      'requests.create',
    ) ?? false

  const canCreateForOthers =
    user?.permissions?.includes(
      'requests.create_for_others',
    ) ?? false

  const canUpdate =
    user?.permissions?.includes(
      'requests.update',
    ) ?? false

  const canSubmit =
    user?.permissions?.includes(
      'requests.submit',
    ) ?? false

  const canDelete =
    user?.permissions?.includes(
      'requests.delete',
    ) ?? false

  const requestsQuery = useQuery({
    queryKey: ['internal-requests', filters],
    queryFn: () =>
      internalRequestService.list(filters),
    enabled: canView,
  })

  const selectedRequestQuery = useQuery({
    queryKey: [
      'internal-request',
      selectedRequestId,
    ],
    queryFn: () =>
      internalRequestService.show(
        selectedRequestId as number,
      ),
    enabled:
      canView && selectedRequestId !== null,
  })

  const requestTypesQuery = useQuery({
    queryKey: ['request-types', 'request-options'],
    queryFn: () =>
      requestTypeService.list({
        page: 1,
        search: '',
        status: 'active',
        per_page: 100,
      }),
    enabled: canView,
  })

  const employeeOptionsQuery = useQuery({
    queryKey: ['employees', 'requester-options'],
    queryFn: () =>
      employeeService.list({
        page: 1,
        search: '',
        status: 'active',
        employment_type: 'all',
        department_id: null,
        sort: 'name_asc',
        per_page: 100,
      }),
    enabled:
      canCreateForOthers && dialogOpen,
  })

  const refreshRequests = async (
    requestId?: number,
  ) => {
    const operations = [
      queryClient.invalidateQueries({
        queryKey: ['internal-requests'],
      }),
      queryClient.invalidateQueries({
        queryKey: ['workflow-tasks'],
      }),
      queryClient.invalidateQueries({
        queryKey: ['dashboard-summary'],
      }),
    ]

    if (requestId !== undefined) {
      operations.push(
        queryClient.invalidateQueries({
          queryKey: [
            'internal-request',
            requestId,
          ],
        }),
      )
    }

    await Promise.all(operations)
  }

  const createMutation = useMutation({
    mutationFn: (
      payload:
        | CreateInternalRequestPayload
        | UpdateInternalRequestPayload,
    ) =>
      internalRequestService.create(
        payload as CreateInternalRequestPayload,
      ),

    onSuccess: async (request) => {
      setSelectedRequestId(request.id)
      await refreshRequests(request.id)
      closeDialog()
    },

    onError: (error) => {
      setMutationError(
        extractErrorMessage(error),
      )
    },
  })

  const updateMutation = useMutation({
    mutationFn: ({
      requestId,
      payload,
    }: {
      requestId: number
      payload:
        | CreateInternalRequestPayload
        | UpdateInternalRequestPayload
    }) =>
      internalRequestService.update(
        requestId,
        payload as UpdateInternalRequestPayload,
      ),

    onSuccess: async (request) => {
      await refreshRequests(request.id)
      closeDialog()
    },

    onError: (error) => {
      setMutationError(
        extractErrorMessage(error),
      )
    },
  })

  const submitMutation = useMutation({
    mutationFn: internalRequestService.submit,

    onSuccess: async (request) => {
      await refreshRequests(request.id)
    },

    onError: (error) => {
      window.alert(extractErrorMessage(error))
    },
  })

  const deleteMutation = useMutation({
    mutationFn: internalRequestService.remove,

    onSuccess: async (_, requestId) => {
      if (selectedRequestId === requestId) {
        setSelectedRequestId(null)
      }

      await refreshRequests()
    },

    onError: (error) => {
      window.alert(extractErrorMessage(error))
    },
  })

  const closeDialog = () => {
    if (
      createMutation.isPending ||
      updateMutation.isPending
    ) {
      return
    }

    setDialogOpen(false)
    setEditingRequest(null)
    setMutationError(null)
  }

  const openCreate = () => {
    setEditingRequest(null)
    setMutationError(null)
    setDialogOpen(true)
  }

  const openEdit = (request: InternalRequest) => {
    setEditingRequest(request)
    setMutationError(null)
    setDialogOpen(true)
  }

  const saveRequest = (
    payload:
      | CreateInternalRequestPayload
      | UpdateInternalRequestPayload,
  ) => {
    if (editingRequest) {
      updateMutation.mutate({
        requestId: editingRequest.id,
        payload,
      })

      return
    }

    createMutation.mutate(payload)
  }

  const submitRequest = (
    request: InternalRequest,
  ) => {
    const confirmed = window.confirm(
      `Submit ${request.reference_number}? The draft will enter its approval workflow.`,
    )

    if (!confirmed) {
      return
    }

    submitMutation.mutate(request.id)
  }

  const removeRequest = (
    request: InternalRequest,
  ) => {
    const confirmed = window.confirm(
      `Delete draft "${request.title}"?`,
    )

    if (!confirmed) {
      return
    }

    deleteMutation.mutate(request.id)
  }

  const applyFilters = (
    event: FormEvent<HTMLFormElement>,
  ) => {
    event.preventDefault()

    setFilters({
      ...draftFilters,
      page: 1,
    })
  }

  if (isUserError) {
    return <Navigate to="/login" replace />
  }

  if (isUserPending) {
    return (
      <main className="grid min-h-screen place-items-center bg-slate-950">
        <LoaderCircle
          size={34}
          className="animate-spin text-sky-400"
        />
      </main>
    )
  }

  if (!canView) {
    return <Navigate to="/dashboard" replace />
  }

  const result = requestsQuery.data
  const requests = result?.data ?? []
  const selectedRequest =
    selectedRequestQuery.data

  return (
    <main className="min-h-screen bg-[#050b18] text-slate-100">
      <header className="sticky top-0 z-30 border-b border-slate-800/80 bg-slate-950/85 backdrop-blur-xl">
        <div className="mx-auto flex min-h-20 max-w-[1700px] items-center justify-between gap-4 px-5 py-4 md:px-8">
          <div className="flex items-center gap-4">
            <button
              type="button"
              onClick={() => navigate('/dashboard')}
              className="grid h-10 w-10 place-items-center rounded-xl border border-slate-800 bg-slate-900/70 text-slate-400 hover:text-sky-300"
            >
              <ArrowLeft size={18} />
            </button>

            <div className="grid h-11 w-11 place-items-center rounded-xl border border-sky-400/15 bg-sky-500/10 text-sky-300">
              <FileText size={22} />
            </div>

            <div>
              <p className="text-xs text-slate-500">
                Workflow operations
              </p>

              <h1 className="font-semibold">
                Internal Requests
              </h1>
            </div>
          </div>

          {canCreate && (
            <button
              type="button"
              onClick={openCreate}
              className="flex h-11 items-center gap-2 rounded-xl bg-sky-500 px-4 text-sm font-semibold text-slate-950 hover:bg-sky-400"
            >
              <Plus size={17} />
              New request
            </button>
          )}
        </div>
      </header>

      <div className="mx-auto max-w-[1700px] p-5 md:p-8">
        <section className="relative overflow-hidden rounded-3xl border border-sky-400/10 bg-gradient-to-br from-blue-950/65 via-slate-950 to-cyan-950/30 p-7">
          <div className="absolute -right-16 -top-20 h-64 w-64 rounded-full bg-sky-500/10 blur-3xl" />

          <div className="relative max-w-3xl">
            <p className="text-xs font-semibold uppercase tracking-[0.16em] text-sky-300">
              Request lifecycle
            </p>

            <h2 className="mt-4 text-3xl font-bold tracking-tight">
              Create, submit and track requests
            </h2>

            <p className="mt-3 max-w-2xl text-sm leading-7 text-slate-400">
              Draft a request, submit it into the selected
              workflow and follow its current step and task
              history.
            </p>
          </div>
        </section>

        <section className="mt-6 rounded-2xl border border-slate-800/80 bg-slate-950/55 p-5">
          <form
            onSubmit={applyFilters}
            className="grid gap-4 xl:grid-cols-[2fr_1fr_1fr_1fr_auto]"
          >
            <label>
              <span className="mb-2 block text-xs text-slate-500">
                Search requests
              </span>

              <div className="flex h-11 items-center gap-3 rounded-xl border border-slate-800 bg-slate-950 px-3">
                <Search
                  size={16}
                  className="text-slate-600"
                />

                <input
                  value={draftFilters.search}
                  onChange={(event) =>
                    setDraftFilters((current) => ({
                      ...current,
                      search: event.target.value,
                    }))
                  }
                  placeholder="Reference, title or description"
                  className="min-w-0 flex-1 bg-transparent text-sm text-slate-200 outline-none placeholder:text-slate-700"
                />
              </div>
            </label>

            <label>
              <span className="mb-2 block text-xs text-slate-500">
                Status
              </span>

              <select
                value={draftFilters.status}
                onChange={(event) =>
                  setDraftFilters((current) => ({
                    ...current,
                    status: event.target
                      .value as InternalRequestFilters['status'],
                  }))
                }
                className="h-11 w-full rounded-xl border border-slate-800 bg-slate-950 px-3 text-sm text-slate-300"
              >
                <option value="all">All statuses</option>
                <option value="draft">Draft</option>
                <option value="in_review">
                  In review
                </option>
                <option value="approved">
                  Approved
                </option>
                <option value="rejected">
                  Rejected
                </option>
                <option value="cancelled">
                  Cancelled
                </option>
              </select>
            </label>

            <label>
              <span className="mb-2 block text-xs text-slate-500">
                Priority
              </span>

              <select
                value={draftFilters.priority}
                onChange={(event) =>
                  setDraftFilters((current) => ({
                    ...current,
                    priority: event.target
                      .value as InternalRequestFilters['priority'],
                  }))
                }
                className="h-11 w-full rounded-xl border border-slate-800 bg-slate-950 px-3 text-sm text-slate-300"
              >
                <option value="all">
                  All priorities
                </option>
                <option value="low">Low</option>
                <option value="normal">
                  Normal
                </option>
                <option value="high">High</option>
                <option value="urgent">
                  Urgent
                </option>
              </select>
            </label>

            <label>
              <span className="mb-2 block text-xs text-slate-500">
                Request type
              </span>

              <select
                value={
                  draftFilters.request_type_id?.toString() ??
                  ''
                }
                onChange={(event) =>
                  setDraftFilters((current) => ({
                    ...current,
                    request_type_id:
                      event.target.value === ''
                        ? null
                        : Number(event.target.value),
                  }))
                }
                className="h-11 w-full rounded-xl border border-slate-800 bg-slate-950 px-3 text-sm text-slate-300"
              >
                <option value="">All types</option>

                {requestTypesQuery.data?.data.map(
                  (requestType) => (
                    <option
                      key={requestType.id}
                      value={requestType.id}
                    >
                      {requestType.name}
                    </option>
                  ),
                )}
              </select>
            </label>

            <div className="flex items-end gap-2">
              <button
                type="submit"
                className="flex h-11 items-center gap-2 rounded-xl bg-sky-500 px-4 text-sm font-semibold text-slate-950 hover:bg-sky-400"
              >
                <Filter size={16} />
                Apply
              </button>

              <button
                type="button"
                onClick={() => {
                  setDraftFilters(initialFilters)
                  setFilters(initialFilters)
                }}
                className="h-11 rounded-xl border border-slate-800 px-4 text-sm text-slate-400"
              >
                Reset
              </button>
            </div>
          </form>
        </section>

        <section className="mt-6 grid gap-6 xl:grid-cols-[430px_minmax(0,1fr)]">
          <aside className="overflow-hidden rounded-2xl border border-slate-800/80 bg-slate-950/55">
            <div className="flex items-center justify-between border-b border-slate-800/80 p-5">
              <div>
                <h3 className="font-semibold">
                  Requests
                </h3>

                <p className="mt-1 text-xs text-slate-600">
                  {result?.meta.total ?? 0} requests found
                </p>
              </div>

              <button
                type="button"
                onClick={() =>
                  void requestsQuery.refetch()
                }
                className="grid h-10 w-10 place-items-center rounded-xl border border-slate-800 text-slate-500 hover:text-sky-300"
              >
                <RefreshCw
                  size={16}
                  className={
                    requestsQuery.isFetching
                      ? 'animate-spin'
                      : ''
                  }
                />
              </button>
            </div>

            <div className="max-h-[760px] space-y-2 overflow-y-auto p-3">
              {requestsQuery.isPending ? (
                <div className="grid min-h-52 place-items-center">
                  <LoaderCircle
                    size={28}
                    className="animate-spin text-sky-400"
                  />
                </div>
              ) : requestsQuery.isError ? (
                <div className="rounded-xl border border-rose-400/20 bg-rose-500/10 p-4 text-sm text-rose-200">
                  Requests could not be loaded.
                </div>
              ) : requests.length > 0 ? (
                requests.map((request) => (
                  <button
                    key={request.id}
                    type="button"
                    onClick={() =>
                      setSelectedRequestId(request.id)
                    }
                    className={`w-full rounded-2xl border p-4 text-left ${
                      selectedRequestId === request.id
                        ? 'border-sky-400/30 bg-sky-500/10'
                        : 'border-slate-800 bg-slate-950/70 hover:border-slate-700'
                    }`}
                  >
                    <div className="flex items-start justify-between gap-3">
                      <div className="min-w-0">
                        <p className="truncate font-medium text-slate-200">
                          {request.title}
                        </p>

                        <p className="mt-1 font-mono text-[11px] text-slate-600">
                          {request.reference_number}
                        </p>
                      </div>

                      <span
                        className={`rounded-full border px-2 py-1 text-[9px] font-semibold uppercase ${statusClasses(
                          request.status,
                        )}`}
                      >
                        {statusLabels[request.status]}
                      </span>
                    </div>

interface ErrorPayload {
  message?: string
  errors?: Record<string, string[]>
}

function errorMessage(error: unknown): string {
  const axiosError = error as AxiosError<ErrorPayload>
  const validation =
    axiosError.response?.data?.errors

  return (
    (validation
      ? Object.values(validation).flat()[0]
      : null) ??
    axiosError.response?.data?.message ??
    'The operation could not be completed.'
  )
}

function formatDate(value: string | null): string {
  if (!value) return 'Not set'

  const date = new Date(value)
  if (Number.isNaN(date.getTime())) return 'Not set'

  return new Intl.DateTimeFormat('en-GB', {
    dateStyle: 'medium',
    timeStyle: 'short',
  }).format(date)
}
